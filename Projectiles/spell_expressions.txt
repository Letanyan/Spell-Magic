arc:  fires multiple projectilesin an arc
	R ; arc radius
	s ; speed
	d ; offset
	
arc-spiral: fires multiple independent oscillating projectiles in an arc
	d ; offset
	s ; speed
	R ; arc radius
	hwave ; projectile horizontal oscillation arc
	vwave ; projectile vertical oscillation arc
	hoff ; projectile horizontal oscillation offset 
	voff ; projectiel vertical oscillation offset
	hmul ; projectile horizontal oscillation rate
	vmul ; projectile vertical oscillation rate
	
bomb: places N projectiles some distance `d*C` away set off after S time in s interval.  
	d ; offset multiplied by C
	S ; minimum wait time
	s ; interval between bombs

bomb-circle: places N projectiles on circle perimeter with radius R in center `d*C`. set off after S time in s interval.  
	d ; offset multiplied by C
	R ; radius of bombs placed
	arc ; arc of the circle
	S ; minimum wait time
	s ; interval between bombs
	
bomb-disc-scatter: places N projectiles in circle with radius R in center `d*C`. set off after S time in s interval.  
	d ; offset multiplied by C
	Rmin ; min radius of bombs placed
	Rmax ; max radius of bombs placed
	arc ; arc of disc
	S ; minimum wait time
	s ; interval between bombs
	
bomb-linear: places N projectiles in line at start `d*C-R` to end `d*C+R`. set off after S time in s interval.  
	d ; offset multiplied by C
	R ; line width of bombs placed
	S ; minimum wait time
	s ; interval between bombs

bomb-sphere-scatter: places N projectiles in sphere with radius R in center `d*C`. set off after S time in s interval.  
	d ; offset multiplied by C
	Rmin ; min radius of bombs placed
	Rmax ; max radius of bombs placed
	arc ; arc of disc
	S ; minimum wait time
	s ; interval between bombs

linear: fires a projectile straight from camera
	s ; speed
	d ; offset
	
linear-arc: fires multiple projectiles in a straight line set in an arc starting point
	R ; arc radius
	d ; offset
	s ; speed
	voff ; vertical offset in radians
	hoff ; horizontal offset in radians
	
linear-arc-spiral: fires multiple independent oscillating projectiles in an straight line set in an arc starting point
	d ; offset
	s ; speed
	R ; arc radius
	hwave ; projectile horizontal oscillation arc
	vwave ; projectile vertical oscillation arc
	hoff ; projectile horizontal oscillation offset 
	voff ; projectiel vertical oscillation offset
	hmul ; projectile horizontal oscillation rate
	vmul ; projectile vertical oscillation rate

linear-flurry: fire multiple shots from a flat circe area 
	d ; offset
	s ; speed
	R ; radius of area for shots fired
	ox, oy, oz ; starting area (offset from camera) * d
	dx, dy, dz ; direction of shots from starting area
	px, py, pz ; area for shots fired
	
linear-turret:
	d ; offset
	ix, iy, iz ; (ix, iy, iz) * n defines line of turret
	s ; speed
	
loop-shot: fires projectiles in an arc trajectory with each projectile spread in an arc
	CC ; distance to travel
	H ; max height
	spread ; horizontal arc
	B ; offset 
	
orbit-xz: projectiles spin around a point
	d ; offset
	s ; rotation speed
	or ; projectile offset
	r ; radius
	
scatter shot: fires multiple projectiles in range defined by harc and varc
	harc ; horizontal arc range
	varc ; vertical arc range
	s ; speed
	
sniper:
	harc ; horizontal xz-arc
	varc ; vertical y-arc
	s ; speed
	
solar-system: fires multiple projectiles that orbit around a center in multiple rings
	r ; base radius
	H ; height
	rs ; speed
	r0 ; offset
	
sphere-linear: fires multiple projectiles from center in multiple directions
	d ; offset
	r ; radius
	dx, dy, dz ; direction
	
swipe: fires projectiles moving along a sphere wall
	d ; base offset
	dz ; z-axis offset for each n-th projectile
	dx ; x-axis angle offset
	dy ; y-axis angle offset
	Dn ; delay for each n-th projectile
	Rx ; x angle to move along
	Ry ; y angle to move along
	 
 	
top-down: fires projectile from top down
	H ; height to start at Top and to travel to Bottom
	Top ; starting height
	Bottom ; distance to travel down

wall:
	s ; speed really small to look static
	h ; y offset
